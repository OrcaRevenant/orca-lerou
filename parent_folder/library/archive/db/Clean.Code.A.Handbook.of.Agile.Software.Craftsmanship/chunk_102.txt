if (f[i]) // if i is uncrossed, cross its multiples.

for (j = 2 * i; j < s; j += i)
f[j] = false; // multiple is not prime
}

// how many primes are there?
int count = 0;
for (i = 0; i < s; i++)
{
if (f[i])
count++; // bump count.
}
int[] primes = new int[count];
// move the primes into the result
for (i = 0, j = 0; i < s; i++)
{
if (f[i])             // if prime
primes[j++] = i;
}
return primes;  // return the primes
}
